# FastAPI Anomaly Detector Endpoint
from fastapi import FastAPI
from pydantic import BaseModel
import joblib

app = FastAPI(title="AegisAI Engine")
model = joblib.load("threat_detector.pkl")

@app.post("/api/v1/telemetry/analyze")
def analyze(packet_rate: float, failed_logins: int):
    score = float(model.predict_proba([[packet_rate, failed_logins]])[0][1])
    return {"risk_score": score, "is_anomaly": score > 0.75}
